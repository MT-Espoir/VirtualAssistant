import json
import os

def load_json_safely(file_path):
    try:
        # Kiểm tra file có tồn tại không
        if not os.path.exists(file_path):
            raise FileNotFoundError(f"File không tồn tại: {file_path}")
        
        # Load và parse JSON
        with open(file_path, 'r', encoding='utf-8') as f:
            data = json.load(f)
            
        return data
    except json.JSONDecodeError as e:
        print(f"Lỗi JSON format: {e}")
        return None
    except FileNotFoundError as e:
        print(f"Lỗi file: {e}")
        return None

def get_nested_value(data, keys, default=None):
    """
    Get value from nested dict using list of keys
    keys: ['daily_learning', 'question_pool', 'personal']

    current = data

import random

def random_select_from_dict(data_dict, exclude_list=[]):
    """
    Randomly select from dictionary, excluding certain items
    """
    if not data_dict:
        return None, None
        
    # Get available keys (excluding already used)
    available_keys = [key for key in data_dict.keys() if key not in exclude_list]
    
    if not available_keys:
        print("Không còn lựa chọn nào khả dụng")
        return None, None
    
    # Random select category
    selected_category = random.choice(available_keys)
    
    # Random select item from that category
    items = data_dict[selected_category]
    if not items:
        return selected_category, None
        
    selected_item = random.choice(items)
    
    return selected_category, selected_item

from datetime import datetime, date
import json

class DateTracker:
    def __init__(self):
        self.last_reset_date = None
        self.daily_data = []
    
    def is_new_day(self):
        """Check if it's a new day since last reset"""
        today = date.today()
        
        if self.last_reset_date is None:
            return True
            
        return today != self.last_reset_date
    
    def reset_daily_data(self):
        """Reset data for new day"""
        if self.is_new_day():
            self.daily_data.clear()
            self.last_reset_date = date.today()
            print(f"Reset data cho ngày mới: {self.last_reset_date}")
            return True
        return False
    
    def add_daily_item(self, item):
        """Add item to today's list"""
        self.reset_daily_data()  # Auto reset if new day
        
        if item not in self.daily_data:
            self.daily_data.append(item)
            return True
        return False
    
    def get_today_summary(self):
        """Get summary of today's activities"""
        return {
            "date": str(date.today()),
            "items_count": len(self.daily_data),
            "items": self.daily_data.copy()
        }

import re
from collections import Counter

class TextAnalyzer:
    def __init__(self):
        # Vietnamese stop words (common words to ignore)
        self.stop_words = {
            'tôi', 'bạn', 'là', 'của', 'và', 'có', 'trong', 'để', 'với', 
            'một', 'này', 'đó', 'được', 'cho', 'từ', 'những', 'các',
            'không', 'khi', 'họ', 'nó', 'sẽ', 'đã', 'bị', 'hay'
        }
    
    def preprocess_text(self, text):
        """Basic text preprocessing"""
        if not text:
            return []
            
        # Lowercase
        text = text.lower()
        
        # Remove special characters but keep Vietnamese chars
        # Keep letters, numbers, spaces
        text = re.sub(r'[^\w\s]', ' ', text)
        
        # Split into words
        words = text.split()
        
        # Remove stop words and short words
        filtered_words = [
            word for word in words 
            if word not in self.stop_words and len(word) > 1
        ]
        
        return filtered_words
    
    def extract_keywords(self, text, top_n=5):
        """Extract most frequent keywords"""
        words = self.preprocess_text(text)
        
        if not words:
            return []
            
        # Count frequency
        word_freq = Counter(words)
        
        # Get top N most common
        return word_freq.most_common(top_n)
    
    def calculate_text_stats(self, text):
        """Calculate basic text statistics"""
        words = self.preprocess_text(text)
        sentences = text.split('.')
        
        return {
            "word_count": len(words),
            "sentence_count": len([s for s in sentences if s.strip()]),
            "avg_word_length": sum(len(w) for w in words) / len(words) if words else 0,
            "unique_words": len(set(words)),
            "top_keywords": self.extract_keywords(text, 3)
        }

# Ví dụ sử dụng

    print(f"  {key}: {value}")

class PersonalityTraitExtractor:
    def __init__(self):
        # Mapping keywords to personality traits
        self.trait_keywords = {
            # Music preferences
            'music_preferences': {
                'pop': 'pop_music',
                'rock': 'rock_music', 
                'jazz': 'jazz_music',
                'classical': 'classical_music',
                'nhạc trẻ': 'young_music',
                'ballad': 'ballad_music'
            },
            
            # Emotional indicators
            'emotional_tone': {
                'vui': 'positive',
                'buồn': 'negative',
                'hạnh phúc': 'positive',
                'stress': 'negative',
                'thích': 'positive_preference',
                'ghét': 'negative_preference',
                'yêu': 'strong_positive'
            },
            
            # Energy level
            'energy_level': {
                'năng động': 'high_energy',
                'tích cực': 'high_energy',
                'mệt': 'low_energy',
                'lười': 'low_energy',
                'thư giãn': 'relaxed'
            },
            
            # Social style
            'social_style': {
                'thích giao tiếp': 'extroverted',
                'nhút nhát': 'introverted',
                'hoạt bát': 'outgoing',
                'yên tĩnh': 'quiet'
            }
        }
    
    def extract_traits(self, text):
        """Extract personality traits from text"""
        text_lower = text.lower()
        extracted_traits = {}
        
        for trait_category, keywords in self.trait_keywords.items():
            found_traits = []
            
            for keyword, trait_value in keywords.items():
                if keyword in text_lower:
                    found_traits.append(trait_value)
            
            if found_traits:
                extracted_traits[trait_category] = found_traits
        
        return extracted_traits
    
    def analyze_sentiment_patterns(self, text):
        """Analyze sentiment patterns in text"""
        positive_words = ['thích', 'yêu', 'vui', 'tuyệt', 'hay', 'tốt']
        negative_words = ['ghét', 'buồn', 'tệ', 'không thích', 'chán']
        
        text_lower = text.lower()
        
        positive_count = sum(1 for word in positive_words if word in text_lower)
        negative_count = sum(1 for word in negative_words if word in text_lower)
        
        if positive_count > negative_count:
            return 'positive'
        elif negative_count > positive_count:
            return 'negative'
        else:
            return 'neutral'

import json
import os
from datetime import datetime

class DataManager:
    def __init__(self, data_dir="user_data"):
        self.data_dir = data_dir
        os.makedirs(data_dir, exist_ok=True)
    
    def save_data(self, filename, data, backup=True):
        """Save data with optional backup"""
        file_path = os.path.join(self.data_dir, filename)
        
        # Create backup if file exists
        if backup and os.path.exists(file_path):
            backup_path = f"{file_path}.backup"
            try:
                os.rename(file_path, backup_path)
            except:
                pass  # Ignore backup errors
        
        try:
            with open(file_path, 'w', encoding='utf-8') as f:
                json.dump(data, f, indent=2, ensure_ascii=False)
            return True
        except Exception as e:
            print(f"Lỗi save data: {e}")
            # Restore backup if save failed
            if backup:
                backup_path = f"{file_path}.backup"
                if os.path.exists(backup_path):
                    os.rename(backup_path, file_path)
            return False
    
    def load_data(self, filename, default=None):
        """Load data with fallback to default"""
        file_path = os.path.join(self.data_dir, filename)
        
        if not os.path.exists(file_path):
            return default if default is not None else {}
        
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                return json.load(f)
        except Exception as e:
            print(f"Lỗi load data: {e}")
            return default if default is not None else {}
    
    def append_log_entry(self, filename, entry):
        """Append entry to JSONL log file"""
        file_path = os.path.join(self.data_dir, filename)
        
        try:
            with open(file_path, 'a', encoding='utf-8') as f:
                f.write(json.dumps(entry, ensure_ascii=False) + '\n')
            return True
        except Exception as e:
            print(f"Lỗi append log: {e}")
            return False

class PersonalityDataMerger:
    def __init__(self):
        self.confidence_weights = {
            'high': 1.0,
            'medium': 0.7,
            'low': 0.4
        }
    
    def merge_traits(self, existing_traits, new_traits, merge_strategy='frequency'):
        """Merge new traits with existing ones"""
        merged = existing_traits.copy()
        
        for trait_category, new_values in new_traits.items():
            if trait_category not in merged:
                merged[trait_category] = []
            
            if merge_strategy == 'frequency':
                # Add new values and count frequency
                merged[trait_category].extend(new_values)
            
            elif merge_strategy == 'unique':
                # Only add if not already present
                for value in new_values:
                    if value not in merged[trait_category]:
                        merged[trait_category].append(value)
            
            elif merge_strategy == 'weighted':
                # Weight recent data higher
                # This is simplified - in real app you'd track timestamps
                merged[trait_category] = new_values + merged[trait_category]
        
        return merged
    
    def calculate_trait_confidence(self, trait_values):
        """Calculate confidence score for trait based on frequency"""
        if not trait_values:
            return 0.0
        
        # Count frequency of each value
        from collections import Counter
        freq = Counter(trait_values)
        
        # Most common value frequency / total
        most_common_freq = freq.most_common(1)[0][1]
        confidence = most_common_freq / len(trait_values)
        
        return confidence
    
    def get_dominant_traits(self, traits, min_confidence=0.6):
        """Get traits that appear with high confidence"""
        dominant = {}
        
        for category, values in traits.items():
            confidence = self.calculate_trait_confidence(values)
            
            if confidence >= min_confidence:
                from collections import Counter
                most_common = Counter(values).most_common(1)[0][0]
                dominant[category] = {
                    'value': most_common,
                    'confidence': confidence
                }
        
        return dominant

class FormalityAnalyzer:
    def __init__(self):
        self.formal_indicators = {
            'pronouns': ['tôi', 'em', 'anh', 'chị', 'quý vị'],
            'politeness': ['ạ', 'dạ', 'kính chào', 'xin lỗi', 'cảm ơn ạ'],
            'formal_words': ['được', 'rất', 'khá', 'hoàn toàn', 'tuyệt đối']
        }
        
        self.casual_indicators = {
            'pronouns': ['mình', 'tao', 'tớ', 'bạn'],
            'casual_words': ['ok', 'yeah', 'nope', 'cool', 'wow'],
            'slang': ['chán', 'ghê', 'ngon', 'đỉnh', 'chill']
        }
    
    def calculate_formality_score(self, text):
        """Calculate formality score (0-1, higher = more formal)"""
        text_lower = text.lower()
        words = text_lower.split()
        
        formal_count = 0
        casual_count = 0
        
        # Count formal indicators
        for category, indicators in self.formal_indicators.items():
            for indicator in indicators:
                formal_count += text_lower.count(indicator)
        
        # Count casual indicators
        for category, indicators in self.casual_indicators.items():
            for indicator in indicators:
                casual_count += text_lower.count(indicator)
        
        total_indicators = formal_count + casual_count
        
        if total_indicators == 0:
            return 0.5  # Neutral if no indicators
        
        formality_score = formal_count / total_indicators
        return formality_score
    
    def analyze_communication_style(self, text_samples):
        """Analyze overall communication style from multiple samples"""
        if not text_samples:
            return {'style': 'unknown', 'confidence': 0.0}
        
        formality_scores = [self.calculate_formality_score(text) for text in text_samples]
        avg_formality = sum(formality_scores) / len(formality_scores)
        
        # Classify style
        if avg_formality > 0.7:
            style = 'formal'
        elif avg_formality < 0.3:
            style = 'casual'
        else:
            style = 'mixed'
        
        # Calculate confidence based on consistency
        score_variance = sum((score - avg_formality) ** 2 for score in formality_scores) / len(formality_scores)
        confidence = 1.0 - min(score_variance * 4, 1.0)  # Lower variance = higher confidence
        
        return {
            'style': style,
            'formality_score': avg_formality,
            'confidence': confidence,
            'sample_count': len(text_samples)
        }

class ResponseLengthAnalyzer:
    def __init__(self):
        self.length_categories = {
            'brief': (1, 5),      # 1-5 words
            'short': (6, 15),     # 6-15 words  
            'medium': (16, 30),   # 16-30 words
            'long': (31, 50),     # 31-50 words
            'very_long': (51, 999) # 51+ words
        }
    
    def categorize_response_length(self, text):
        """Categorize response by word count"""
        word_count = len(text.split())
        
        for category, (min_words, max_words) in self.length_categories.items():
            if min_words <= word_count <= max_words:
                return category
        
        return 'very_long'  # Default for very long responses
    
    def analyze_length_preferences(self, response_texts):
        """Analyze user's preferred response length"""
        if not response_texts:
            return {'preferred_length': 'unknown', 'distribution': {}}
        
        # Categorize all responses
        categories = [self.categorize_response_length(text) for text in response_texts]
        
        # Count distribution
        from collections import Counter
        distribution = Counter(categories)
        
        # Find most common length
        preferred_length = distribution.most_common(1)[0][0]
        
        # Calculate percentages
        total_responses = len(response_texts)
        distribution_percent = {
            category: (count / total_responses) * 100 
            for category, count in distribution.items()
        }
        
        return {
            'preferred_length': preferred_length,
            'distribution': distribution_percent,
            'total_responses': total_responses,
            'avg_word_count': sum(len(text.split()) for text in response_texts) / total_responses
        }
    
    def recommend_response_length(self, user_preference):
        """Recommend response length based on user preference"""
        recommendations = {
            'brief': "Keep responses short and direct (5-10 words)",
            'short': "Use concise responses (10-20 words)", 
            'medium': "Provide moderate detail (20-35 words)",
            'long': "Give detailed responses (35-50 words)",
            'very_long': "Provide comprehensive responses (50+ words)"
        }
        
        return recommendations.get(user_preference, "Adapt to context")

class ComprehensiveUserAnalyzer:
    def __init__(self):
        self.trait_extractor = PersonalityTraitExtractor()
        self.formality_analyzer = FormalityAnalyzer()
        self.length_analyzer = ResponseLengthAnalyzer()
        self.text_analyzer = TextAnalyzer()
    
    def comprehensive_analysis(self, user_responses):
        """Perform comprehensive analysis of user"""
        if not user_responses:
            return self.get_default_profile()
        
        # Extract personality traits
        all_traits = {}
        for response in user_responses:
            traits = self.trait_extractor.extract_traits(response)
            all_traits = self.merge_trait_data(all_traits, traits)
        
        # Analyze communication style
        style_analysis = self.formality_analyzer.analyze_communication_style(user_responses)
        
        # Analyze length preferences
        length_analysis = self.length_analyzer.analyze_length_preferences(user_responses)
        
        # Calculate engagement metrics
        engagement = self.calculate_engagement_metrics(user_responses)
        
        return {
            'personality_traits': all_traits,
            'communication_style': style_analysis,
            'response_preferences': length_analysis,
            'engagement_metrics': engagement,
            'analysis_date': datetime.now().isoformat(),
            'sample_size': len(user_responses)
        }
    
    def merge_trait_data(self, existing, new_traits):
        """Merge trait data from multiple responses"""
        merged = existing.copy()
        
        for category, values in new_traits.items():
            if category not in merged:
                merged[category] = []
            merged[category].extend(values)
        
        return merged
    
    def calculate_engagement_metrics(self, responses):
        """Calculate user engagement metrics"""
        if not responses:
            return {}
        
        total_words = sum(len(response.split()) for response in responses)
        avg_response_length = total_words / len(responses)
        
        # Count questions (engagement indicator)
        question_count = sum(response.count('?') for response in responses)
        
        # Calculate vocabulary richness
        all_words = []
        for response in responses:
            all_words.extend(self.text_analyzer.preprocess_text(response))
        
        unique_words = len(set(all_words))
        vocabulary_richness = unique_words / len(all_words) if all_words else 0
        
        return {
            'avg_response_length': avg_response_length,
            'total_responses': len(responses),
            'questions_asked': question_count,
            'vocabulary_richness': vocabulary_richness,

class AdaptiveResponseGenerator:
    def __init__(self):
        self.response_templates = {
            'formal': {
                'acknowledgment': "Tôi hiểu ý kiến của bạn về {topic}",
                'question': "Bạn có thể chia sẻ thêm về {topic} được không ạ?",
                'information': "Theo như tôi biết về {topic}, {information}"
            },
            'casual': {
                'acknowledgment': "Mình hiểu rồi về {topic}",
                'question': "Kể thêm về {topic} đi bạn?",
                'information': "Về {topic} thì {information}"
            },
            'friendly': {
                'acknowledgment': "Thật thú vị khi bạn nói về {topic}!",
                'question': "Bạn có câu chuyện nào khác về {topic} không?",
                'information': "Hay nhỉ! Về {topic}, mình biết là {information}"
            }
        }
    
    def generate_adaptive_response(self, user_profile, context, response_type='acknowledgment'):
        """Generate response adapted to user's style"""
        # Determine style to use
        comm_style = user_profile.get('communication_style', {}).get('style', 'mixed')
        
        if comm_style == 'formal':
            style_key = 'formal'
        elif comm_style == 'casual':
            style_key = 'casual'
        else:
            style_key = 'friendly'  # Default friendly tone
        
        # Get template
        template = self.response_templates[style_key].get(response_type, "Tôi hiểu.")
        
        # Fill in context
        response = template.format(**context)
        
        # Adjust length based on user preference
        response = self.adjust_response_length(response, user_profile)
        
        # Add personality elements
        response = self.add_personality_elements(response, user_profile)
        
        return response
    
    def adjust_response_length(self, response, user_profile):
        """Adjust response length based on user preference"""
        preferred_length = user_profile.get('response_preferences', {}).get('preferred_length', 'medium')
        
        if preferred_length == 'brief':
            # Shorten response
            words = response.split()
            if len(words) > 8:
                response = ' '.join(words[:8]) + "..."
        
        elif preferred_length in ['long', 'very_long']:
            # Add more detail (simplified example)
            if not response.endswith('.'):
                response += "."
            response += " Bạn có muốn tìm hiểu thêm không?"
        
        return response
    
    def add_personality_elements(self, response, user_profile):
        """Add elements based on user's personality"""
        traits = user_profile.get('personality_traits', {})
        
        # If user likes music, occasionally mention it
        if 'music_preferences' in traits and len(response.split()) > 5:
            music_prefs = traits['music_preferences']
            if music_prefs and 'pop' in music_prefs:
                # Randomly add music reference (simplified)
                import random
                if random.random() < 0.3:  # 30% chance
                    response += " (Như bạn thích nhạc pop, phải không?)"
        
        return response

# Ví dụ sử dụng
